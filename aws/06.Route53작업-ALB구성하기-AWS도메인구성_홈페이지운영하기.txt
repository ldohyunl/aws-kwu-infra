# 1. 변수 선언
# CLB_DNS 변수에 로드밸런서 주소를 넣어주세요.
CLB_DNS="kwu-nginx-clb-366373775.ap-northeast-2.elb.amazonaws.com"

# 2. 한 줄에 한 회차씩 완벽하게 매핑되는 숏 가이드 스크립트
for i in {1..10}; do
    # curl 결과를 변수에 담아 중복을 제거한 후 한 줄로만 뽑아냅니다.
    RESULT=$(curl -s --connect-timeout 2 "http://$CLB_DNS" | grep -o -E "Nginx-2A|Nginx-2C" | head -n 1)
    
    # 템플릿에 맞추어 깔끔하게 출력
    echo "[$i회차 요청 결과] 👉 $RESULT"
    sleep 0.5
done



[모니터링] v2
* Cloudshell 에서
# 1. CLB DNS 주소 정의 (프로토콜과 슬래시를 제외한 순수 DNS만 입력)
CLB_DNS="kwu-nginx-clb-366373775.ap-northeast-2.elb.amazonaws.com"

echo "=========================================================="
echo "🔴  KWU CLB 10회 한정 레드/블루 라우팅 테스트 가동 🔵"
echo "=========================================================="

# 10회 반복 구조 실행
for i in {1..10}; do
    # 📌 [CLB 보정] 명시적으로 http:// 프로토콜을 앞에 붙여서 정밀 노킹
    RAW_RESP=$(curl -s --connect-timeout 2 "http://$CLB_DNS")
    CHECK_SERVER=$(echo "$RAW_RESP" | grep -o -E "Nginx-2A|Nginx-2C" | head -n 1)
    CURRENT_TIME=$(date +'%H:%M:%S')

    # 2. 추출된 서버명 및 회차 정보 매핑 출력
    if [ "$CHECK_SERVER" == "Nginx-2A" ]; then
        echo -e "[$CURRENT_TIME] [${i}회차] 🚀 CLB ➡️ \e[1;31m[Nginx-2A (RED Zone) 응답]\e[0m"
    elif [ "$CHECK_SERVER" == "Nginx-2C" ]; then
        echo -e "[$CURRENT_TIME] [${i}회차] 🚀 CLB ➡️ \e[1;34m[Nginx-2C (BLUE Zone) 응답]\e[0m"
    else
        echo -e "[$CURRENT_TIME] [${i}회차] ⚠️  \e[1;33m[CLB 연동 지연 또는 타임아웃 발생]\e[0m"
    fi

    # 다음 요청까지 0.5초 대기
    sleep 0.5
done

echo "=========================================================="
echo "✅ CLB 테스트가 완료되었습니다. 프롬프트로 복귀합니다."
echo "=========================================================="

2. Application LoadBalancer
2-1) ALB 보안그룹 생성
[EC2] - [보안그룹] - [보안그룹 생성] 클릭


보안 그룹 이름과 설명은 ‘KWU-PRD-VPC-ALB-SG ’로 입력하며
VPC는 이전에 생성한 ‘KWU-PRD-VPC’로 지정
인바운드와 아웃바운드 규칙은 생성 이후 지정할 것이므로 [보안 그룹 생성] 클릭

생성 시 다음과 같이 결과가 나옴
인바운드 규칙 설정을 위해 [인바운드 규칙] - [인바운드 규칙 편집] 클릭

[규칙 추가] 클릭 후 아래 표를 참고하여 인바운드 규칙 설정 후 [규칙 저장] 클릭
HTTP	Anywhere IPv4	0.0.0.0/0
HTTPS	Anywhere IPv4	0.0.0.0/0



2-2) ALB 대상그룹 생성
[EC2] - [대상 그룹] - [대상 그룹 생성] 클릭
 * ALB는 대상그룹부터 만들어줘야 함. 
 * 나중에 '대상그룹'을 지우려면, 연결된 로드밸런서 부터 지워야 함. 
 
 > [기본 구성]은 ‘인스턴스’로 선택 (단. Container 환경일 경우 대상 유형 선택 : IP 주소 선택)
 > [대상 그룹 이름]은 관리대장 참고하여 ‘ KWU-PRD-VPC-ALB-TG ’로 입력
 > 프로토콜과 IP 주소 유형은 Default값 사용 (80, HTTP 로만 생성)
 > [VPC]는 이전에 생성한 ‘KWU-PRD-VPC’ 선택 후 
 > 맨밑 [다음] 클릭 
 > [대상 등록]에서 이전에 생성한 EC2 모두 선택 후 (BASTION  제외)
 > [아래에 보류 중인 것으로 포함] 클릭 
 > 맨밑 다음
 > '검토 및 생성' 페이지 내용 확인 후 [대상 그룹 생성] 클릭   
 > ‘Unused’ 상태로 있음 (로드 밸런서 생성이 활성화 되면 바뀌게 된다.)



2-3) ALB 생성
[EC2] - [로드밸런서] - [로드 밸런서 생성] 클릭

등록된 대상 이 unused 여서 ALB를 등록해야 사용가능함. 

‘Application Load Balancer’ 클릭 후 [생성] 클릭
 > 로드 밸런서 이름은 ‘KWU-PRD-VPC-ALB’로 지정
 > 체계 : 인터넷 경계
 > 로드 밸런서 IP 주소 유형 : IPv4
 > 네트워크 매핑 : VPC는 생성한 ‘KWU-PRD-VPC’로 지정
 > IP 풀 : 선택X 
 > 가용영역 클릭 후 서브넷 ‘KWU-PRD-VPC-NGINX-PUB-2A’와 ‘KWU-PRD-VPC-NGINX-PUB-2C’ 선택
 > 보안그룹은 이전에 생성한 ‘KWU-PRD-VPC-ALB-SG’ 선택  

* 리스너는 아래 표 참고하여 80먼저 하고 [리스너 추가] 버튼 클릭 후 HTTPS 추가하기
 HTTP  80  KWU-PRD-VPC-ALB-TG    (HTTP)
 HTTPS 443 KWU-PRD-VPC-ALB-TG    (HTTP)
 
- 리스너  
 > HTTP  80 
 > 기본 작업 , 라우팅 액션 - 대상 그룹으로 전달
 > 우리가 만든 대상 그룹 선택 
 
- 리스너 추가 
 > HTTPS   443
 > 기본 작업 , 라우팅 액션 - 대상 그룹으로 전달
 > 우리가 만든 대상 그룹 선택 
 
 
‘보안 리스너 설정’의 인증서(ACM에서)에는 생성한 인증서 지정 후 [로드밸런서 생성] 클릭

다음과 같이 생성된 결과 확인 가능 
 > 프로비저닝이 끝나야 health 로 변경되어짐. 
 > 이때 대상 그룹의 상태에 변경이 나타남! ‘INITIAL’ 상태로 전환 → ‘HEALTHY’로 추후 전환
 > 나중에 활성 상태로 바뀌며 DNS 주소로 결과 확인 가능  
 > 따라서, 아래 Route 53 수정 부터 해줘야 함.

# 1. ALB 도메인 주소 정의 (프로토콜 포함하여 깔끔하게 선언)
ALB_DNS="https://graycoldsky.store"

echo "=========================================================="
echo "🔴  KWU ALB 실시간 레드/블루 모니터링 가동 (종료: Ctrl+C) 🔵"
echo "=========================================================="

while true; do
    # 📌 [교정 포인트 1] ALB_DNS 앞에 $를 붙이고, 쌍따옴표 내부 프로토콜 꼬임 방지
    RAW_RESP=$(curl -s --connect-timeout 2 "$ALB_DNS")
    
    CHECK_SERVER=$(echo "$RAW_RESP" | grep -o -E "Nginx-2A|Nginx-2C" | head -n 1)
    CURRENT_TIME=$(date +'%H:%M:%S')

    # 2. 추출된 서버명에 따라 터미널에 리얼 컬러 출력 매핑
    if [ "$CHECK_SERVER" == "Nginx-2A" ]; then
        echo -e "[$CURRENT_TIME] 🚀 ALB 트래픽 라우팅 ➡️ \e[1;31m[Nginx-2A (RED Zone) 응답 성공]\e[0m"
    elif [ "$CHECK_SERVER" == "Nginx-2C" ]; then
        echo -e "[$CURRENT_TIME] 🚀 ALB 트래픽 라우팅 ➡️ \e[1;34m[Nginx-2C (BLUE Zone) 응답 성공]\e[0m"
    else
        # 📌 만약 여전히 노란색이 뜬다면, CloudShell에서 해당 도메인으로 curl 자체가 안 되는 네트워크 상황입니다.
        echo -e "[$CURRENT_TIME] ⚠️  \e[1;33m[ALB 연동 지연 또는 타임아웃 발생]\e[0m"
    fi

    sleep 0.8
done
